option(TRAFFIC_BUILD_UI "Build the Windows ImGui live viewer" ON)
if(TRAFFIC_BUILD_UI AND WIN32)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/libs/imgui/imgui.cpp")
    message(FATAL_ERROR "ImGui missing: run git submodule update --init libs/imgui or configure -DTRAFFIC_BUILD_UI=OFF")
  endif()
  add_library(traffic_imgui STATIC
    libs/imgui/imgui.cpp libs/imgui/imgui_draw.cpp
    libs/imgui/imgui_tables.cpp libs/imgui/imgui_widgets.cpp
    libs/imgui/backends/imgui_impl_win32.cpp
    libs/imgui/backends/imgui_impl_opengl3.cpp)
  target_include_directories(traffic_imgui PUBLIC libs/imgui libs/imgui/backends)
  target_link_libraries(traffic_imgui PUBLIC opengl32 gdi32 dwmapi)
  add_executable(traffic_ui src/UiMain.cpp src/SimulationView.cpp)
  set(TRAFFIC_UI_OUTPUT_NAME "traffic_capacity" CACHE STRING "Viewer executable name (use another name while a viewer is running)")
  set_target_properties(traffic_ui PROPERTIES OUTPUT_NAME "${TRAFFIC_UI_OUTPUT_NAME}")
  target_compile_definitions(traffic_ui PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_definitions(traffic_ui PRIVATE TRAFFIC_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(traffic_ui PRIVATE traffic traffic_imgui)
  add_custom_command(TARGET traffic_ui POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:traffic_ui>/examples"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/examples/map.png" "$<TARGET_FILE_DIR:traffic_ui>/examples/map.png"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/examples/annotated-map.png" "$<TARGET_FILE_DIR:traffic_ui>/examples/annotated-map.png"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/examples/curved-mask.png" "$<TARGET_FILE_DIR:traffic_ui>/examples/curved-mask.png")
endif()
